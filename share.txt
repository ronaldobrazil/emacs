   /// @brief 装置の稼働状態
   enum class OperationalStatus : uint32_t {
       Unknown   = 0,
       Up        = 1,
       Degraded  = 2,
       Down      = 3
   };

   constexpr std::string_view toString(OperationalStatus status) noexcept {
       switch (status) {
           case OperationalStatus::Up:       return "UP";
           case OperationalStatus::Degraded: return "DEGRADED";
           case OperationalStatus::Down:     return "DOWN";
           default:                          return "UNKNOWN";
       }
   }

   /// @brief 稼働時間 (sysUpTime: 1/100秒単位 = centiseconds)
   class DeviceUptime {
   public:
       constexpr explicit DeviceUptime(uint64_t centisecs = 0) noexcept : centisecs_(centisecs) {}

       [[nodiscard]] constexpr uint64_t centiseconds() const noexcept { return centisecs_; }
       [[nodiscard]] constexpr uint64_t totalSeconds() const noexcept { return centisecs_ / 100; }

       [[nodiscard]] std::string format() const {
           const uint64_t s = totalSeconds();
           const uint64_t days  = s / 86400;
           const uint64_t hours = (s % 86400) / 3600;
           const uint64_t mins  = (s % 3600) / 60;
           const uint64_t secs  = s % 60;
           return std::format("{}d {:02}h {:02}m {:02}s ({}.{:02}s)",
                              days, hours, mins, secs, s, centisecs_ % 100);
       }

   private:
       uint64_t centisecs_{0};
   };

   /// @brief 定期ポーリング ユースケース (GETBULK)
   class PollDeviceUseCase {
   public:
       PollDeviceUseCase(std::shared_ptr<ISnmpClient> snmp_client,
                         std::shared_ptr<ISharedMemoryGateway> shm_gateway)
           : snmp_client_(std::move(snmp_client)), shm_gateway_(std::move(shm_gateway)) {}

       void execute(std::string_view target_ip, Domain::DeviceSnapshot& snapshot) {
           // sysUpTime (.1.3.6.1.2.1.1.3.0) および system MIB (.1.3.6.1.2.1.1) を GETBULK
           const Domain::Oid system_mib_root{".1.3.6.1.2.1.1"};
           auto vars_opt = snmp_client_->getBulk(target_ip, system_mib_root, 0, 10);

           snapshot.ip_address = std::string(target_ip);
           snapshot.last_poll_time = std::chrono::system_clock::now();

           if (!vars_opt || vars_opt->empty()) {
               snapshot.status = Domain::OperationalStatus::Down;
               shm_gateway_->writeSnapshot(snapshot);
               return;
           }

           snapshot.status = Domain::OperationalStatus::Up;

           for (const auto& vb : *vars_opt) {
               // sysDescr.0 (.1.3.6.1.2.1.1.1.0)
               if (vb.oid.toString().starts_with(".1.3.6.1.2.1.1.1.0") ||
                   vb.oid.toString().starts_with("1.3.6.1.2.1.1.1.0")) {
                   if (auto* str = std::get_if<std::string>(&vb.value)) {
                       snapshot.sys_descr = *str;
                   }
               }
               // sysUpTimeInstance (.1.3.6.1.2.1.1.3.0)
               if (vb.oid.toString().starts_with(".1.3.6.1.2.1.1.3.0") ||
                   vb.oid.toString().starts_with("1.3.6.1.2.1.1.3.0")) {
                   if (auto* ut = std::get_if<uint64_t>(&vb.value)) {
                       snapshot.uptime = Domain::DeviceUptime(*ut);
                   } else if (auto* it = std::get_if<int64_t>(&vb.value)) {
                       snapshot.uptime = Domain::DeviceUptime(static_cast<uint64_t>(*it));
                   }
               }
           }

           shm_gateway_->writeSnapshot(snapshot);
       }

   private:
       std::shared_ptr<ISnmpClient> snmp_client_;
       std::shared_ptr<ISharedMemoryGateway> shm_gateway_;
   };

   /// @brief Trap 処理ユースケース (変数バインディングの解析)
   class ProcessTrapUseCase {
   public:
       explicit ProcessTrapUseCase(std::shared_ptr<ISharedMemoryGateway> shm_gateway)
           : shm_gateway_(std::move(shm_gateway)) {}

       void execute(const Domain::TrapEvent& trap, Domain::DeviceSnapshot& snapshot) {
           if (trap.source_ip != snapshot.ip_address) {
               return; // 対象外の装置からのTrapはスキップ
           }

           snapshot.last_trap_time = trap.received_at;
           snapshot.last_trap_oid = trap.trap_oid.toString();
           snapshot.uptime = trap.uptime;

           // 変数バインディングから詳細情報を集約
           std::string detail_summary;
           for (const auto& vb : trap.bindings) {
               std::string val_str;
               std::visit([&val_str](const auto& v) {
                   using T = std::decay_t<decltype(v)>;
                   if constexpr (std::is_same_v<T, std::string>) val_str = v;
                   else if constexpr (std::is_same_v<T, int64_t>) val_str = std::to_string(v);
                   else if constexpr (std::is_same_v<T, uint64_t>) val_str = std::to_string(v);
                   else if constexpr (std::is_same_v<T, Domain::Oid>) val_str = v.toString();
                   else val_str = "null";
               }, vb.value);

               detail_summary += std::format("[{}: {}={}] ", vb.type_name, vb.oid.toString(), val_str);
           }
           snapshot.last_trap_detail = detail_summary;

           // 代表的な RFC1907 / IF-MIB 標準Trapの判定
           const auto& toid = trap.trap_oid.toString();
           if (toid.ends_with(".1.3.6.1.6.3.1.1.5.3") || toid.find("linkDown") != std::string::npos) {
               snapshot.status = Domain::OperationalStatus::Degraded;
           } else if (toid.ends_with(".1.3.6.1.6.3.1.1.5.4") || toid.find("linkUp") != std::string::npos) {
               snapshot.status = Domain::OperationalStatus::Up;
           } else if (toid.ends_with(".1.3.6.1.6.3.1.1.5.1") || toid.find("coldStart") != std::string::npos) {
               snapshot.status = Domain::OperationalStatus::Up;
           }

           shm_gateway_->writeSnapshot(snapshot);
       }

   private:
       std::shared_ptr<ISharedMemoryGateway> shm_gateway_;
   };

   // 共有メモリ上のバイナリ配置構造体 (POD / Standard Layout)
   struct alignas(64) ShmDeviceStatusPayload {
       pthread_mutex_t mutex;
       uint64_t sequence_id;              // 整合性確認用シーケンス
       char target_ip[46];                // IPv4/IPv6文字列
       uint32_t operational_state;        // Domain::OperationalStatus
       uint64_t uptime_centiseconds;      // sysUpTime
       int64_t last_poll_epoch_ms;        // 最終ポーリング日時
       int64_t last_trap_epoch_ms;        // 最終Trap受信日時
       char sys_descr[256];               // システム説明
       char last_trap_oid[128];           // 直近のTrap OID
       char last_trap_detail[512];        // VarBind詳細サマリ
   };

   /// @brief Linux POSIX 共有メモリ Gateway 実装
   class PosixSharedMemoryGateway : public UseCases::ISharedMemoryGateway {
   public:
       explicit PosixSharedMemoryGateway(std::string shm_name)
           : shm_name_(std::move(shm_name)) {

           // 共有メモリの作成またはオープン
           fd_ = ::shm_open(shm_name_.c_str(), O_CREAT | O_RDWR, 0666);
           if (fd_ == -1) {
               throw std::runtime_error("Failed to shm_open: " + std::string(strerror(errno)));
           }

           if (::ftruncate(fd_, sizeof(ShmDeviceStatusPayload)) == -1) {
               ::close(fd_);
               throw std::runtime_error("Failed to ftruncate shm: " + std::string(strerror(errno)));
           }

           void* addr = ::mmap(nullptr, sizeof(ShmDeviceStatusPayload),
                               PROT_READ | PROT_WRITE, MAP_SHARED, fd_, 0);
           if (addr == MAP_FAILED) {
               ::close(fd_);
               throw std::runtime_error("Failed to mmap shm: " + std::string(strerror(errno)));
           }

           payload_ = static_cast<ShmDeviceStatusPayload*>(addr);

           // プロセス間共有 Robust Mutex の初期化
           pthread_mutexattr_t mattr;
           ::pthread_mutexattr_init(&mattr);
           ::pthread_mutexattr_setpshared(&mattr, PTHREAD_PROCESS_SHARED);
           ::pthread_mutexattr_setrobust(&mattr, PTHREAD_MUTEX_ROBUST);
           ::pthread_mutex_init(&payload_->mutex, &mattr);
           ::pthread_mutexattr_destroy(&mattr);
       }

       ~PosixSharedMemoryGateway() override {
           if (payload_) {
               ::munmap(payload_, sizeof(ShmDeviceStatusPayload));
           }
           if (fd_ != -1) {
               ::close(fd_);
           }
           // 必要に応じて shm_unlink(shm_name_.c_str());
       }

       void writeSnapshot(const Domain::DeviceSnapshot& snapshot) override {
           if (!payload_) return;

           ::pthread_mutex_lock(&payload_->mutex);

           payload_->sequence_id++;
           std::strncpy(payload_->target_ip, snapshot.ip_address.c_str(), sizeof(payload_->target_ip) - 1);
           payload_->operational_state = static_cast<uint32_t>(snapshot.status);
           payload_->uptime_centiseconds = snapshot.uptime.centiseconds();

           auto to_ms = [](std::chrono::system_clock::time_point tp) -> int64_t {
               return std::chrono::duration_cast<std::chrono::milliseconds>(tp.time_since_epoch()).count();
           };

           payload_->last_poll_epoch_ms = to_ms(snapshot.last_poll_time);
           payload_->last_trap_epoch_ms = to_ms(snapshot.last_trap_time);

           std::strncpy(payload_->sys_descr, snapshot.sys_descr.c_str(), sizeof(payload_->sys_descr) - 1);
           std::strncpy(payload_->last_trap_oid, snapshot.last_trap_oid.c_str(), sizeof(payload_->last_trap_oid) - 1);
           std::strncpy(payload_->last_trap_detail, snapshot.last_trap_detail.c_str(),
   sizeof(payload_->last_trap_detail) - 1);

           ::pthread_mutex_unlock(&payload_->mutex);
       }

   private:
       std::string shm_name_;
       int fd_{-1};
       ShmDeviceStatusPayload* payload_{nullptr};
   };

   /// @brief Net-SNMP 内部データ型から Domain::VarBind へのコンバータ
   class NetSnmpValueConverter {
   public:
       static Domain::VarBind convert(const variable_list* var) {
           Domain::VarBind vb;

           // OIDの変換
           char oid_buf[MAX_OID_LEN * 4 + 1];
           snprint_objid(oid_buf, sizeof(oid_buf), var->name, var->name_length);
           vb.oid = Domain::Oid(oid_buf);

           // 型判別と値の格納
           switch (var->type) {
               case ASN_INTEGER:
                   vb.type_name = "INTEGER";
                   vb.value = static_cast<int64_t>(*var->val.integer);
                   break;
               case ASN_TIMETICKS:
                   vb.type_name = "TIMETICKS";
                   vb.value = static_cast<uint64_t>(*var->val.integer);
                   break;
               case ASN_COUNTER:
               case ASN_GAUGE:
                   vb.type_name = "COUNTER/GAUGE";
                   vb.value = static_cast<uint64_t>(*var->val.integer);
                   break;
               case ASN_OCTET_STR: {
                   vb.type_name = "OCTET_STR";
                   std::string str(reinterpret_cast<const char*>(var->val.string), var->val_len);
                   vb.value = str;
                   break;
               }
               case ASN_OBJECT_ID: {
                   vb.type_name = "OBJECT_ID";
                   char sub_oid_buf[MAX_OID_LEN * 4 + 1];
                   snprint_objid(sub_oid_buf, sizeof(sub_oid_buf), var->val.objid, var->val_len / sizeof(oid));
                   vb.value = Domain::Oid(sub_oid_buf);
                   break;
               }
               default:
                   vb.type_name = std::format("TYPE_{}", var->type);
                   vb.value = std::monostate{};
                   break;
           }
           return vb;
       }
   };

   /// @brief Net-SNMP ライブラリによる GETBULK クライアント実装
   class NetSnmpClient : public UseCases::ISnmpClient {
   public:
       explicit NetSnmpClient(std::string community = "public")
           : community_(std::move(community)) {
           static std::once_flag init_flag;
           std::call_once(init_flag, []() {
               ::init_snmp("snmp_v2c_monitor_client");
           });
       }

       std::optional<std::vector<Domain::VarBind>> getBulk(
           std::string_view host,
           const Domain::Oid& root_oid,
           int non_repeaters,
           int max_repetitions) override {

           netsnmp_session session{};
           ::snmp_sess_init(&session);
           session.version = SNMP_VERSION_2c;
           session.peername = const_cast<char*>(host.data());
           session.community = reinterpret_cast<u_char*>(const_cast<char*>(community_.c_str()));
           session.community_len = community_.length();
           session.retries = 2;
           session.timeout = 1000000; // 1秒 (マイクロ秒単位)

           netsnmp_session* ss = ::snmp_open(&session);
           if (!ss) {
               return std::nullopt;
           }

           // PDU の作成: SNMP_MSG_GETBULK
           netsnmp_pdu* pdu = ::snmp_pdu_create(SNMP_MSG_GETBULK);
           pdu->non_repeaters = non_repeaters;
           pdu->max_repetitions = max_repetitions;

           oid anOID[MAX_OID_LEN];
           size_t anOID_len = MAX_OID_LEN;
           if (!::snmp_parse_oid(root_oid.toString().c_str(), anOID, &anOID_len)) {
               ::snmp_free_pdu(pdu);
               ::snmp_close(ss);
               return std::nullopt;
           }

           ::snmp_add_null_var(pdu, anOID, anOID_len);

           netsnmp_pdu* response = nullptr;
           int status = ::snmp_synch_response(ss, pdu, &response);

           if (status != STAT_SUCCESS || !response || response->errstat != SNMP_ERR_NOERROR) {
               if (response) ::snmp_free_pdu(response);
               ::snmp_close(ss);
               return std::nullopt;
           }

           std::vector<Domain::VarBind> results;
           for (variable_list* vars = response->variables; vars != nullptr; vars = vars->next_variable) {
               // MIB Viewの終端判定
               if (vars->type == SNMP_ENDOFMIBVIEW ||
                   vars->type == SNMP_NOSUCHOBJECT ||
                   vars->type == SNMP_NOSUCHINSTANCE) {
                   break;
               }
               results.push_back(NetSnmpValueConverter::convert(vars));
           }

           ::snmp_free_pdu(response);
           ::snmp_close(ss);
           return results;
       }

   private:
       std::string community_;
   };

   /// @brief Net-SNMP による非同期 Trap 受信デーモン実装
   class NetSnmpTrapDaemon : public UseCases::ITrapReceiver {
   public:
       explicit NetSnmpTrapDaemon(int listen_port = 1162) : listen_port_(listen_port) {}

       ~NetSnmpTrapDaemon() override {
           stop();
       }

       void start(TrapHandler handler) override {
           handler_ = std::move(handler);
           worker_ = std::jthread([this](std::stop_token stop_token) {
               runReceiverLoop(stop_token);
           });
       }

       void stop() override {
           if (worker_.joinable()) {
               worker_.request_stop();
               worker_.join();
           }
       }

   private:
       static int trapCallback(int op, netsnmp_session* session, int reqid,
                               netsnmp_pdu* pdu, void* magic) {
           auto* self = static_cast<NetSnmpTrapDaemon*>(magic);
           if (op == NETSNMP_CALLBACK_OP_RECEIVED_MESSAGE && pdu) {
               self->onTrapPdu(pdu);
           }
           return 1;
       }

       void onTrapPdu(netsnmp_pdu* pdu) {
           if (pdu->command != SNMP_MSG_TRAP2 && pdu->command != SNMP_MSG_INFORM) {
               return;
           }

           Domain::TrapEvent event;
           event.received_at = std::chrono::system_clock::now();

           // 送信元IPアドレスの取得
           sockaddr_in* addr = reinterpret_cast<sockaddr_in*>(pdu->transport_data);
           if (addr) {
               char ip_str[INET_ADDRSTRLEN];
               ::inet_ntop(AF_INET, &(addr->sin_addr), ip_str, sizeof(ip_str));
               event.source_ip = ip_str;
           }

           // 変数バインディングの解析
           for (variable_list* vars = pdu->variables; vars != nullptr; vars = vars->next_variable) {
               auto vb = NetSnmpValueConverter::convert(vars);

               // RFC 3416: 1番目のVarBindは sysUpTime.0
               if (vb.oid.toString().find("sysUpTime") != std::string::npos ||
                   vb.oid.toString().find(".1.3.6.1.2.1.1.3.0") != std::string::npos) {
                   if (auto* ut = std::get_if<uint64_t>(&vb.value)) {
                       event.uptime = Domain::DeviceUptime(*ut);
                   }
               }
               // 2番目のVarBindは snmpTrapOID.0
               else if (vb.oid.toString().find("snmpTrapOID") != std::string::npos ||
                        vb.oid.toString().find(".1.3.6.1.6.3.1.1.4.1.0") != std::string::npos) {
                   if (auto* toid = std::get_if<Domain::Oid>(&vb.value)) {
                       event.trap_oid = *toid;
                   }
               } else {
                   event.bindings.push_back(std::move(vb));
               }
           }

           if (handler_) {
               handler_(event);
           }
       }

       void runReceiverLoop(std::stop_token stop_token) {
           netsnmp_session session{};
           ::snmp_sess_init(&session);
           session.version = SNMP_VERSION_2c;
           session.peername = const_cast<char*>(std::format("udp:{}", listen_port_).c_str());
           session.callback = trapCallback;
           session.callback_magic = this;

           netsnmp_session* ss = ::snmp_open(&session);
           if (!ss) {
               std::cerr << std::format("[TrapDaemon] Failed to bind to port {} (Need sudo for 1162?)\n", listen_port_);
               return;
           }

           while (!stop_token.stop_requested()) {
               int numfds = 0;
               fd_set fdset;
               timeval timeout{.tv_sec = 0, .tv_usec = 200000}; // 200msポーリング
               int block = 0;

               FD_ZERO(&fdset);
               ::snmp_select_info(&numfds, &fdset, &timeout, &block);
               int rc = ::select(numfds, &fdset, nullptr, nullptr, &timeout);

               if (rc > 0) {
                   ::snmp_read(&fdset);
               } else if (rc == 0) {
                   ::snmp_timeout();
               }
           }

           ::snmp_close(ss);
       }

       int listen_port_;
       TrapHandler handler_;
       std::jthread worker_;
   };

   } // namespace Infrastructure

   // =============================================================================
   // 4. Interface Adapters Layer (Coordinator & Composition Root)
   // =============================================================================
   namespace Adapters {

   class MonitoringCoordinator {
   public:
       MonitoringCoordinator(
           std::string target_ip,
           std::chrono::milliseconds poll_interval,
           std::shared_ptr<UseCases::ISnmpClient> snmp_client,
           std::shared_ptr<UseCases::ITrapReceiver> trap_receiver,
           std::shared_ptr<UseCases::ISharedMemoryGateway> shm_gateway)
           : target_ip_(std::move(target_ip)),
             poll_interval_(poll_interval),
             trap_receiver_(std::move(trap_receiver)),
             poll_use_case_(snmp_client, shm_gateway),
             trap_use_case_(shm_gateway) {

           snapshot_.ip_address = target_ip_;
       }

       void start() {
           // 1. Trap 受信デーモンの起動とハンドラ登録
           trap_receiver_->start([this](const Domain::TrapEvent& trap) {
               std::lock_guard lock(snapshot_mutex_);
               std::cout << std::format("\033[1;35m[TRAP RECEIVED]\033[0m From: {} | TrapOID: {}\n",
                                        trap.source_ip, trap.trap_oid.toString());
               trap_use_case_.execute(trap, snapshot_);
           });

           // 2. 定期 GETBULK ポーリングスレッドの起動
           poller_thread_ = std::jthread([this](std::stop_token stop_token) {
               std::cout << std::format("[Poller] Starting GETBULK monitoring for {}\n", target_ip_);
               while (!stop_token.stop_requested()) {
                   {
                       std::lock_guard lock(snapshot_mutex_);
                       poll_use_case_.execute(target_ip_, snapshot_);
                       printLocalStatus(snapshot_);
                   }

                   // 中断可能な待機
                   std::unique_lock<std::mutex> cv_lock(cv_mutex_);
                   cv_.wait_for(cv_lock, poll_interval_, [&stop_token]() {
                       return stop_token.stop_requested();
                   });
               }
           });
       }

       void stop() {
           if (poller_thread_.joinable()) {
               poller_thread_.request_stop();
               cv_.notify_all();
               poller_thread_.join();
           }
           trap_receiver_->stop();
       }

   private:
       static void printLocalStatus(const Domain::DeviceSnapshot& s) {
           std::string_view color = (s.status == Domain::OperationalStatus::Up) ? "\033[1;32m" : "\033[1;31m";
           std::cout << std::format("[STATUS] Host: {:<15} | State: {}{:<8}\033[0m | Uptime: {}\n",
                                    s.ip_address, color, Domain::toString(s.status), s.uptime.format());
       }

       std::string target_ip_;
       std::chrono::milliseconds poll_interval_;
       std::shared_ptr<UseCases::ITrapReceiver> trap_receiver_;
       UseCases::PollDeviceUseCase poll_use_case_;
       UseCases::ProcessTrapUseCase trap_use_case_;

       Domain::DeviceSnapshot snapshot_;
       std::mutex snapshot_mutex_;
       std::condition_variable cv_;
       std::mutex cv_mutex_;
       std::jthread poller_thread_;
   };

   } // namespace Adapters

   // =============================================================================
   // Application Entry Point
   // =============================================================================
   int main() {
       using namespace std::chrono_literals;

       constexpr const char* TARGET_IP = "192.168.1.241";
       constexpr const char* SHM_NAME  = "/snmp_monitor_192_168_1_241";
       constexpr int TRAP_PORT         = 1162; // 一般ユーザー権限でテスト時は 1162 等に変更可

       std::cout << "========================================================\n";
       std::cout << "  SNMP v2c Monitor (Clean Architecture & C++20)\n";
       std::cout << std::format("  Target: {} | SHM: {} | TrapPort: {}\n", TARGET_IP, SHM_NAME, TRAP_PORT);
       std::cout << "========================================================\n";

       try {
           // 1. Framework / Infrastructure Layer のインスタンス生成
           auto snmp_client   = std::make_shared<Infrastructure::NetSnmpClient>("public");
           auto trap_receiver = std::make_shared<Infrastructure::NetSnmpTrapDaemon>(TRAP_PORT);
           auto shm_gateway   = std::make_shared<Infrastructure::PosixSharedMemoryGateway>(SHM_NAME);

           // 2. Interface Adapters Layer の起動 (2秒間隔ポーリング)
           Adapters::MonitoringCoordinator coordinator(
               TARGET_IP, 2000ms, snmp_client, trap_receiver, shm_gateway
           );

           coordinator.start();

           // 30秒間モニタリングを実行 (Ctrl+C で中断可能)
           std::cout << "Monitoring running. Press Enter to stop...\n";
           std::cin.get();

           std::cout << "Stopping monitoring...\n";
           coordinator.stop();

       } catch (const std::exception& ex) {
           std::cerr << "Fatal Error: " << ex.what() << '\n';
           return 1;
       }

       return 0;
   }
